const PLANILHA_ID = 'COLE_AQUI_O_ID_DA_PLANILHA'; const NOME_ABA = 'Confirmações'; const CABECALHO = ['Data/hora', 'Nome', 'WhatsApp', 'Vem?', 'Adultos', 'Crianças', 'Total', 'Acompanhantes', 'Recado']; function doPost(e) { try { salvar_((e && e.parameter) || {}); return resposta_({ ok: true }); } catch (err) { return resposta_({ ok: false, erro: String(err) }); } } function doGet(e) { if (e && e.parameter && e.parameter.teste) { try { salvar_({ nome: 'TESTE (pode apagar)', whatsapp: '(21) 00000-0000', presenca: 'Sim', adultos: 1, criancas: 0, total: 1 }); return ContentService.createTextOutput('OK ✔ Linha de teste gravada na aba "' + NOME_ABA + '".'); } catch (err) { return ContentService.createTextOutput('ERRO: ' + err); } } return ContentService.createTextOutput('Confirmação do Benício funcionando ✔'); } function salvar_(p) { const lock = LockService.getScriptLock(); lock.waitLock(10000); try { const aba = pegarAba_(); aba.appendRow([ new Date(), limpar_(p.nome), limpar_(p.whatsapp), limpar_(p.presenca), Number(p.adultos) || 0, Number(p.criancas) || 0, Number(p.total) || 0, limpar_(p.acompanhantes), limpar_(p.recado) ]); } finally { lock.releaseLock(); } } function pegarAba_() { const ss = (PLANILHA_ID && PLANILHA_ID.indexOf('COLE_AQUI') === -1) ? SpreadsheetApp.openById(PLANILHA_ID) : SpreadsheetApp.getActiveSpreadsheet(); if (!ss) throw new Error('Planilha não encontrada: cole o ID da planilha em PLANILHA_ID.'); let aba = ss.getSheetByName(NOME_ABA); if (!aba) { const primeira = ss.getSheets()[0]; if (primeira.getLastRow() === 0) { aba = primeira; aba.setName(NOME_ABA); } else aba = ss.insertSheet(NOME_ABA, 0); } if (aba.getLastRow() === 0) { aba.appendRow(CABECALHO); aba.getRange(1, 1, 1, CABECALHO.length) .setFontWeight('bold').setBackground('#f6e27a').setFontColor('#141414'); aba.setFrozenRows(1); aba.getRange('A:A').setNumberFormat('dd/MM/yyyy HH:mm'); aba.getRange('K1').setValue('Total confirmados').setFontWeight('bold'); aba.getRange('K2').setFormula('=SUMIF(D2:D;"Sim";G2:G)'); } return aba; } function resposta_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function limpar_(v) { v = String(v == null ? '' : v).trim().slice(0, 1000); return /^[=+\-@]/.test(v) ? "'" + v : v; }
